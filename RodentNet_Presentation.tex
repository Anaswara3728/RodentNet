\begin{frame}{Methodology: The Zero-Sum Mathematical Mechanism}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \textbf{1. 40-Bit Prime Galois Discretization:}
      \begin{equation*}
        p = 2^{40} - 87 = 1{,}099{,}511{,}627{,}689
      \end{equation*}
      \begin{itemize}[itemsep=1pt]
        \item Continuous weights $w \in [-10, 10]$ quantized with scale $S = 10^6$: $x = \lfloor w \cdot S \rceil \bmod p$.
        \item Sum of 50 clients cannot exceed $5 \times 10^8 \ll p/2 \approx 5.5 \times 10^{11}$ ($\to$ zero modular overflow).
      \end{itemize}
      \vspace{0.2cm}
      \textbf{2. Anti-Symmetric Pairwise Blinding:}
      \begin{itemize}[itemsep=1pt]
        \item Ephemeral seeds: $K_{i,j}^{(r)} = \text{SHAKE-256}(s_{i,j} \parallel r, 32)$.
        \item If $i < j$: add $+\text{PRG}(K_{i,j}^{(r)})$.
        \item If $i > j$: subtract $-\text{PRG}(K_{j,i}^{(r)})$.
      \end{itemize}
    \end{column}
    \begin{column}{0.5\textwidth}
      \textbf{3. Exact Algebraic Cancellation:}
      \begin{equation*}
        \widetilde{\mathbf{x}}_i^{(r)} = \mathbf{x}_i^{(r)} + \mathbf{b}_i + \sum_{j > i} \mathbf{m}_{i,j}^{(r)} - \sum_{j < i} \mathbf{m}_{j,i}^{(r)} \pmod p
      \end{equation*}
      When summed across all surviving clients $\mathcal{S}$:
      \begin{equation*}
        \sum_{i \in \mathcal{S}} \widetilde{\mathbf{x}}_i^{(r)} \equiv \sum_{i \in \mathcal{S}} \mathbf{x}_i^{(r)} + \sum_{i \in \mathcal{S}} \mathbf{b}_i + \mathbf{0} \pmod p
      \end{equation*}
      \begin{exampleblock}{Provable Equivalence Floor}
        An isolated single-round test on identical model updates confirmed that Plain FedAvg and Secured MPC aggregation yield weights identical within:
        \begin{equation*}
          \max |\Delta\theta| = (2.42 \pm 0.30) \times 10^{-7}
        \end{equation*}
        strictly bounded by the $10^{-6}$ quantization scale.
      \end{exampleblock}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Methodology: Dynamic Dropout Resilience (Model B)}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \textbf{The Dropout Recovery Protocol:}
      \begin{enumerate}[itemsep=2pt]
        \item \textbf{Setup Sharing:} Each client splits its ephemeral pairwise seeds $K_{i,j}^{(r)}$ into $(t, N)$ Shamir shares over $\text{GF}(257)$.
        \item \textbf{Dropout Event:} Client $k$ crashes or disconnects. Its pairwise masks held by survivors are stranded.
        \item \textbf{Reconstruction:} Surviving peers upload their shares of $K_{k,j}^{(r)}$ to the server.
        \item \textbf{Lagrange Interpolation:} The server reconstructs the seeds in $\text{GF}(257)$:
        \begin{equation*}
          f(0) = \sum_{m=1}^t y_m \prod_{\ell \ne m} \frac{-\ell}{m - \ell} \pmod{257}
        \end{equation*}
        \item \textbf{Noise Stripping:} The server generates the stranded masks and subtracts them from the accumulator.
      \end{enumerate}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{Why Model B (Pairwise Sharing)?}
        \footnotesize
        In classical SecAgg (Model A), sharing a client's private key allows deriving all its pairwise masks. However, \textbf{asymmetric post-quantum KEMs (Kyber768) prevent this}: knowing client $u$'s private key does not reveal shared secrets without peer ciphertexts.
        
        \textbf{Model B} directly shares the pairwise seeds, scaling cohort-wide as $N^2(N-1)/2$ shares while keeping client uplink bounded to $O(N)$.
      \end{block}
      \begin{block}{Forward Secrecy Guarantee}
        \footnotesize
        Deriving seeds via non-invertible SHAKE-256 ratchets ensures that reconstructing $K_{k,j}^{(r)}$ reveals \textbf{zero mathematical information} about past rounds $r' < r$ or future root keys.
      \end{block}
    \end{column}
  \end{columns}
\end{frame}
